\section{Summary}

\begin{frame}{Summary}
  \begin{itemize}
	\item
		\emph{As a kid, as my father drove across the desert, I heard melodies and chords within the wind outside the car.}
	\item
		This is my attempt to evoke such harmonies from various sounds.
	\item
		The basic idea is to convolve a source sound with the impulse response of a specific chord, \emph{i.e.} of the chord's \emph{pitch-classes}, on \emph{every} octave.
	\end{itemize}
\end{frame}

\begin{frame}{Related Techniques?}
  \begin{itemize}
	\item
		Constant-Q filter bank, \emph{iff} its filters are centered on the pitch-classes.
	\item
	\emph{Not closely related}: Gabor/Morland wavelet, but its cells are not centered on pitch-classes -- not a good fit.
	\item \emph{Even less related}: Comb filter, but that's periodic in linear frequency; this is logarithmic in frequency period.
\end{itemize}
\end{frame}

\section{Motivation}

\begin{frame}{Musical Motivation}
  \begin{itemize}
		\item
			There has been a disjunction between "non-tonal" (loosely speaking) electroacoustic music and 'non-tonal" (loosely speaking) electroacoustic music.
		\item 
		    This goes back to the roots of computer music in the 1950s and 1960s, a period dominated by technological utopianism and musical serialism.
		\item "Non-tonal" has implied a focus on \textgreek{>ako\'usmata}; "tonal" implies a focus on pitch relations abstracted from actual sounds.
		\item This particular algorithm works at the \emph{boundary} of timbre and tonality.
\end{itemize}
\end{frame}

\section{How It Works}

\begin{frame}{The Mathematics}

For each pitch class $c$, the corresponding filter responses form an octave-related family of scaled kernels.

\[
	H_C(f)
	=
	\sum_{c\in C}\sum_n H_{c,n}(f),
	\qquad
	H_{c,n}(f)
	\simeq
	H_c\!\left(2^{-n}f\right).
\]

\end{frame}

\begin{frame}{The Mathematics (continued)}
In log-frequency coordinates
\[
	u=\log_2 f,
\]
octave transposition becomes translation by one unit, and the total chordal response is approximately periodic with period one octave:
\[
	H_C(u+1)\simeq H_C(u).
\]
Each octave therefore contains $K$ peaks, whose positions within the octave are determined by the $K$ pitch classes of the chord. The chord functions as the unit cell of a periodic filter in log-frequency space.
\end{frame}

\begin{frame}[fragile]{Algorithm (1/2): partials}
\footnotesize
\begin{verbatim}
harmony_convolver_scaled(x, T, g_imp, g_dirac, PC[1..5]):
  T     = clamp(abs(T), 0.01, 1.0)
  n_max = T * 440
  N     = max(2, round(T * sr))
  // PC x octave partials + loudness weights
  freqs = [];  weights = []
  for midi = 0..127:
    if (midi mod 12) not in PCs: continue
    f = cpsmidinn(midi)
    if f < 20 or f > 0.95*Nyquist: continue
    fp = max(f, 150)            // rumble shelf
    w  = R_A(1000) / R_A(fp)    // inverse A-weight
    w  = min(w, 10^(6/20))      // cap boost +6 dB
    append f to freqs
    append w to weights
  M = length(freqs)
  h = zeros(N)
\end{verbatim}
\end{frame}

\begin{frame}[fragile]{Algorithm (2/2): grain, mix, convolve}
\footnotesize
\begin{verbatim}
  // Build harmonic grain
  if M > 0:
    for n = 0..N-1:
      t = n / sr;  s = 0
      for i = 0..M-1:
        f  = freqs[i]
        Tf = min(T, n_max / f)  // shorter at high f
        if t < Tf:
          hann = 0.5-0.5*cos(2*pi*t/Tf)
          s = s + weights[i]*hann*sin(2*pi*f*t)
      h[n] = (s / M) * g_imp
  // Dry+wet in one IR, then unit L2
  h[0] = h[0] + g_dirac
  h    = h / L2norm(h)
  return ftconv(x, h)
\end{verbatim}
\end{frame}
