\section{Summary}

\begin{frame}{Summary}
  \begin{itemize}
	\item
		\emph{When I was a boy, as my father drove across the desert at night, I
		heard melodies and chords in the wind outside the car.}
	\item
		This is my attempt to evoke such harmonies from various sounds.
	\item
		The basic idea is to convolve a source sound with the impulse response of a
		specific chord, \emph{i.e.} of the chord's \emph{pitch-classes}, on
		\emph{every} octave.
	\end{itemize}
\end{frame}

\begin{frame}{Related Techniques?}
  \begin{itemize}
	\item
		Constant-Q filter bank, \emph{iff} its filters are centered on the
		pitch-classes.
	\item
	\emph{Not closely related}: Gabor/Morlet wavelet, but its cells are not
	centered on pitch-classes -- not a good fit.
	\item \emph{Even less related}: Comb filter, but that's periodic in linear
	Hz; this is periodic in log frequency (one octave).
\end{itemize}
\end{frame}

\section{Why Do This?}

\begin{frame}{Musical Motivation}
  \begin{itemize}
		\item
			There has been a disjunction between "non-tonal" (loosely speaking)
			electroacoustic practice and tonal practice oriented toward pitch
			relations abstracted from actual sounds.
		\item 
		    This goes back to the roots of computer music in the 1950s and 1960s, a
		    period dominated by technological Utopianism and musical serialism.
		\item "Non-tonal" has implied a focus on \emph{acousmata}; "tonal" implies a
		focus on pitch relations abstracted from actual sounds.
		\item This particular algorithm works at the \emph{boundary} of timbre and
		tonality.
\end{itemize}
\end{frame}

\section{How It Works}

\begin{frame}{Mathematics}

For each pitch class $c$ of a chord $C$, the corresponding filter responses
form an octave-related family of scaled kernels.

\[
	H_C(f)
	=
	\sum_{c\in C}\sum_n H_{c,n}(f),
	\qquad
	H_{c,n}(f)
	\simeq
	H_c\!\left(2^{-n}f\right).
\]

Grain length
\[
T(f)=\min\bigl(T,\,T\cdot 440/f\bigr)
\]
is constant-$Q$ above $440\,\mathrm{Hz}$, so that scaling is exact there;
below the pivot, duration is capped at $T$.

\end{frame}

\begin{frame}{Mathematics (continued)}
In log-frequency coordinates
\[
	u=\log_2 f,
\]
octave transposition becomes translation by one unit, and the total chordal
response is approximately periodic with period one octave:
\[
	H_C(u+1)\simeq H_C(u).
\]
Each octave therefore contains $K$ peaks, whose positions within the octave are
determined by the $K$ pitch classes of the chord. The chord functions as the
unit cell of a periodic filter in log-frequency space.
\end{frame}

\begin{frame}{Levelling Octaves}
Above the pivot, grains shorten with frequency, so equal amplitude is not
equal presence. Two stages:
\begin{enumerate}
\item \textbf{Equal energy.} Sources are broadband, so by Parseval the power
      that partial $k$ contributes is
      \[
        P_k = S_0\!\int\! \bigl|G_k(f)\bigr|^2 df = S_0\!\int\! g_k(t)^2 dt =
        S_0 E_k ,
      \]
      hence $A(f)\propto T(f)^{-1/2}$.
\item \textbf{Inverse A-weighting}, converting equal power into equal loudness.
\end{enumerate}
{\small
Loudness tracks power, not spectral density: every partial occupies under one
auditory filter ($\le 0.74$ ERB). Normalizing the window \emph{integral}
instead fixes peak gain while the bandwidth $\approx 1/T(f)$ widens, tilting
the lattice bright by ${\sim}8\,\mathrm{dB}$.}
\end{frame}

\begin{frame}{Latent, Not Added}
The aim is that the harmony seem \emph{latent} in the source and evoked from
it, not added to it.
\begin{itemize}
\item Convolution serves this by construction: $Y(f)=H(f)X(f)$ vanishes
      wherever the source has no energy, so the lattice can only expose pitch
      content the source already contains.
\item Energy persisting after the source falls silent betrays the effect by
      sounding like a separate resonating object.
\item The causal window and the cap on $T(f)$ are therefore fusion devices,
      keeping the harmonic response inside the events of the source.
\end{itemize}
{\small On a $50\,\mathrm{ms}$ impulse response driven by a $3\,\mathrm{Hz}$
impulse train, this design leaves under $3\,\%$ of each period's energy
outside the interval the dry source occupies; uncapped or symmetrically
windowed grains displace a third or more.}
\end{frame}

\begin{frame}[fragile]{Algorithm (1/2): partials}
\scriptsize
\begin{verbatim}
harmony_convolver_scaled(x, T, g_imp, g_dirac, PC[1..5]):
  T     = clamp(abs(T), 0.01, 1.5)
  n_max = T * 440
  N     = max(2, round(T * sr))
  // degrees reduce mod 12; negatives unused
  PCs   = { p mod 12 : p in PC, p >= 0 }
  // PC x octave partials + loudness weights
  freqs = [];  weights = [];  Tf = []
  for midi = 0..127:
    if (midi mod 12) not in PCs: continue
    f = cpsmidinn(midi)
    if f < 20 or f > 0.95*Nyquist: continue
    fp = max(f, 150)            // rumble shelf
    w  = R_A(1000) / R_A(fp)    // inverse A-weight
    w  = min(w, 10^(6/20))      // cap boost +6 dB
    append f to freqs
    append w to weights
    append min(T, n_max / f) to Tf
  M = length(freqs)
\end{verbatim}
\end{frame}

\begin{frame}[fragile]{Algorithm (2/2): equal energy, mix}
\scriptsize
\begin{verbatim}
  // Equal energy per partial, total energy held fixed
  // (default opcode; no ERB correction)
  amp = []
  for i = 0..M-1:  amp[i] = sqrt(T / Tf[i])
  c = sqrt(sum(Tf) / sum(amp^2 * Tf))
  for i = 0..M-1:  amp[i] = c * amp[i]
  h = zeros(N)
  for n = 0..N-1:
    t = n / sr;  s = 0
    for i = 0..M-1:
      if t < Tf[i]:
        env = 0.5+0.5*cos(pi*t/Tf[i])  // causal, 1 -> 0
        s = s + weights[i]*amp[i]*env*sin(2*pi*freqs[i]*t)
    h[n] = (s / M) * g_imp
  // Dry+wet in one IR, then unit L2
  h[0] = h[0] + g_dirac
  h    = h / L2norm(h)
  return ftconv(x, h)
\end{verbatim}
\end{frame}
